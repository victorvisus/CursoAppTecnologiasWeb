CREATE TABLE equipo(
	id INT(11) AUTO_INCREMENT PRIMARY KEY,
	nombre VARCHAR(100)
);

CREATE TABLE simpatia(
	id INT AUTO_INCREMENT PRIMARY KEY,
	id_persona INT,
	id_equipo INT,
	CONSTRAINT fk_persona
		FOREIGN KEY (id_persona)
		REFERENCES persona(id)
		ON DELETE CASCADE
		ON UPDATE CASCADE,
	CONSTRAINT fk_equipo
		FOREIGN KEY (id_equipo)
		REFERENCES equipo(id)
		ON DELETE CASCADE
		ON UPDATE CASCADE
);
ALTER TABLE simpatia MODIFY id_persona INT NOT NULL;
ALTER TABLE simpatia MODIFY id_equipo INT NOT NULL;
